cmake_minimum_required(VERSION 3.12)
project(molecular_orbital_simulation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(OpenMP)
file(GLOB LIB_SRC ${CMAKE_SOURCE_DIR}/src/*.cpp)
list(REMOVE_ITEM LIB_SRC ${CMAKE_SOURCE_DIR}/src/main.cpp)
add_library(molorb_core STATIC ${LIB_SRC})
target_include_directories(molorb_core PUBLIC src)
target_compile_options(molorb_core PRIVATE -Wall -Wextra)
if(OpenMP_CXX_FOUND)
  target_link_libraries(molorb_core PUBLIC OpenMP::OpenMP_CXX)
endif()

add_executable(molorb src/main.cpp)
target_link_libraries(molorb molorb_core)

enable_testing()
add_executable(test_runner tests/test_main.cpp)
target_link_libraries(test_runner molorb_core)
add_test(NAME regression COMMAND test_runner)
